async function loadContent() {
    try {
        const response = await fetch('data.json');
        const data = await response.json();

        // Populate Hero
        document.getElementById('hero-title').innerText = data.hero.title;
        document.getElementById('hero-desc').innerText = data.hero.subtitle;
        document.getElementById('hero-btn').innerText = data.hero.cta;

        // Populate Products
        const productContainer = document.getElementById('product-list');
        data.products.forEach(item => {
            const section = document.createElement('section');
            section.className = `section-container ${item.layout === 'right' ? 'row-right' : 'row-left'}`;
            
            section.innerHTML = `
                <div class="content-box">
                    <h2>${item.title}</h2>
                    <p>${item.description}</p>
                    <button class="btn-outline">${item.button}</button>
                </div>
                <div class="image-placeholder"></div>
            `;
            productContainer.appendChild(section);
        });
    } catch (error) {
        console.error("Error fetching the JSON data:", error);
    }
}

loadContent();
